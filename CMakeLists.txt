cmake_minimum_required(VERSION 3.22)
project(Turi VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_program(VELOZ_CLANG NAMES clang-23 clang HINTS "C:/Program Files/LLVM/bin")
add_executable(veloz src/main.cpp src/agent.cpp src/format.cpp src/driver.cpp src/generics.cpp src/parser.cpp src/semantics.cpp src/native.cpp)
if(VELOZ_CLANG)
  target_compile_definitions(veloz PRIVATE VELOZ_CLANG_PATH="${VELOZ_CLANG}")
endif()
if(WIN32)
  target_link_libraries(veloz PRIVATE shell32)
endif()
if(MSVC)
  target_compile_options(veloz PRIVATE /utf-8 /W4)
else()
  target_compile_options(veloz PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_executable(veloz_semantics_tests tests/semantics_runner.cpp src/parser.cpp src/semantics.cpp src/generics.cpp)
  target_include_directories(veloz_semantics_tests PRIVATE src)
  if(MSVC)
    target_compile_options(veloz_semantics_tests PRIVATE /utf-8)
  endif()
  add_test(NAME semantics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/semantics.py" "$<TARGET_FILE:veloz_semantics_tests>")
  add_test(NAME agent COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/agent.py" "$<TARGET_FILE:veloz>")
  add_test(NAME core COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/core.py" "$<TARGET_FILE:veloz>")
  if(VELOZ_CLANG)
    add_test(NAME differential COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/differential.py" "$<TARGET_FILE:veloz>")
    add_test(NAME native_edges COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/native_edges.py" "$<TARGET_FILE:veloz>")
    add_test(NAME arrays COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/arrays.py" "$<TARGET_FILE:veloz>")
  endif()
endif()
